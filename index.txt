const db = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);
const $ = id => document.getElementById(id);
let user = null, accounts = [], activeId = null;

function say(el, text, ok){ el.textContent = text || ""; el.classList.toggle("ok", !!ok); }

async function route(){
  const { data } = await db.auth.getSession();
  user = data.session ? data.session.user : null;
  $("auth").hidden = !!user;
  $("app").hidden = !user;
  if (user){ $("who").textContent = user.email; await loadAccounts(); }
}

$("btnLogin").onclick = async () => {
  const { error } = await db.auth.signInWithPassword({ email: $("email").value.trim(), password: $("pass").value });
  if (error) say($("authMsg"), "Não foi possível entrar: " + error.message); else route();
};
$("btnSignup").onclick = async () => {
  const { data, error } = await db.auth.signUp({ email: $("email").value.trim(), password: $("pass").value });
  if (error) return say($("authMsg"), "Não foi possível criar a conta: " + error.message);
  if (data.session) route(); else say($("authMsg"), "Conta criada. Confirme pelo link enviado ao seu e-mail e depois entre.", true);
};
$("btnReset").onclick = async () => {
  const email = $("email").value.trim();
  if (!email) return say($("authMsg"), "Digite seu e-mail para receber o link de nova senha.");
  const { error } = await db.auth.resetPasswordForEmail(email, { redirectTo: location.href.split("#")[0] });
  say($("authMsg"), error ? "Não foi possível enviar: " + error.message : "Enviamos o link de nova senha para o seu e-mail.", !error);
};
$("btnLogout").onclick = async () => { await db.auth.signOut(); activeId = null; route(); };

async function loadAccounts(){
  const { data, error } = await db.from("game_accounts").select("id,name").order("created_at");
  if (error) return say($("appMsg"), "Erro ao carregar as contas: " + error.message);
  accounts = data;
  activeId = localStorage.getItem("activeAccount");
  if (!accounts.some(a => a.id === activeId)) activeId = accounts.length ? accounts[0].id : null;
  render();
}

function render(){
  const ul = $("list"); ul.innerHTML = "";
  if (!accounts.length) ul.innerHTML = '<li>Você ainda não tem contas do jogo. Adicione a primeira acima.</li>';
  accounts.forEach(a => {
    const li = document.createElement("li");
    if (a.id === activeId) li.className = "active";
    const name = document.createElement("span"); name.className = "name"; name.textContent = a.name;
    const use = btn("Usar", () => { activeId = a.id; localStorage.setItem("activeAccount", a.id); render(); });
    const ren = btn("Renomear", () => rename(a));
    const del = btn("Excluir", () => remove(a)); del.classList.add("danger");
    [use, ren, del].forEach(b => b.classList.add("link"));
    li.append(name, use, ren, del); ul.append(li);
  });
  const a = accounts.find(x => x.id === activeId);
  $("current").textContent = a ? "Conta ativa: " + a.name : "";
  showCards();
}
function btn(t, fn){ const b = document.createElement("button"); b.textContent = t; b.onclick = fn; return b; }

$("btnAdd").onclick = async () => {
  const name = $("newName").value.trim();
  if (!name) return say($("appMsg"), "Digite um nome para a conta do jogo.");
  const { error } = await db.from("game_accounts").insert({ name });
  if (error) return say($("appMsg"), "Não foi possível adicionar: " + error.message);
  $("newName").value = ""; say($("appMsg"), ""); loadAccounts();
};
async function rename(a){
  const name = (prompt("Novo nome da conta do jogo:", a.name) || "").trim();
  if (!name || name === a.name) return;
  const { error } = await db.from("game_accounts").update({ name }).eq("id", a.id);
  if (error) say($("appMsg"), "Não foi possível renomear: " + error.message); else loadAccounts();
}
async function remove(a){
  if (!confirm("Excluir “" + a.name + "” e toda a coleção dela? Isso não pode ser desfeito.")) return;
  const { error } = await db.from("game_accounts").delete().eq("id", a.id);
  if (error) say($("appMsg"), "Não foi possível excluir: " + error.message); else loadAccounts();
}

// ---- Cartas (dados abertos da comunidade, versão fixa) ----
const DATA = "https://cdn.jsdelivr.net/npm/pokemon-tcg-pocket-database@2.10.0/dist/";
const RAR = { C:"Comum", U:"Incomum", R:"Rara", RR:"Dupla rara", SR:"Super rara", AR:"Arte rara", SAR:"Arte especial", IM:"Imersiva", UR:"Coroa" };
let allCards = null, owned = {}, wished = {}, loadedKey = null;

async function showCards(){
  $("cardsPanel").hidden = !activeId;
  if (!activeId) return;
  try {
    if (!allCards){
      const [c, s] = await Promise.all([
        fetch(DATA + "cards.no-image.min.json").then(r => r.json()),
        fetch(DATA + "sets.json").then(r => r.json())
      ]);
      allCards = c;
      const sets = Object.values(s).flat().sort((a, b) => b.releaseDate.localeCompare(a.releaseDate));
      $("setSel").innerHTML = "";
      sets.forEach(x => { const o = document.createElement("option"); o.value = x.code; o.textContent = x.code + " · " + (x.name.pt || x.name.en); $("setSel").append(o); });
    }
    loadOwned();
  } catch (e) { say($("cardsMsg"), "Não foi possível carregar a lista de cartas. Verifique a conexão e recarregue a página."); }
}
async function loadOwned(force){
  const set = $("setSel").value, key = activeId + "|" + set;
  if (!force && key === loadedKey) return renderCards();
  const [a, w] = await Promise.all([
    db.from("collection").select("card_id,quantity").eq("account_id", activeId).like("card_id", set + "-%"),
    db.from("wishlist").select("card_id").eq("account_id", activeId).like("card_id", set + "-%")
  ]);
  if (a.error || w.error) return say($("cardsMsg"), "Erro ao carregar a coleção: " + (a.error || w.error).message);
  owned = {}; a.data.forEach(r => owned[r.card_id] = r.quantity);
  wished = {}; w.data.forEach(r => wished[r.card_id] = true);
  loadedKey = key; say($("cardsMsg"), ""); renderCards();
}
function renderCards(){
  const set = $("setSel").value, term = $("q").value.trim().toLowerCase(), view = $("view").value;
  const cards = allCards.filter(c => c.set === set).sort((a, b) => a.number - b.number);
  const have = cards.filter(c => owned[set + "-" + c.number] > 0).length;
  $("progress").textContent = have + " de " + cards.length + " cartas";
  $("barFill").style.width = (cards.length ? have / cards.length * 100 : 0) + "%";
  const ul = $("cards"); ul.innerHTML = "";
  cards.filter(c => {
    const id = set + "-" + c.number;
    return (!term || c.name.toLowerCase().includes(term)) &&
      (view === "all" || (view === "missing" && !owned[id]) || (view === "wish" && wished[id]) || (view === "dupes" && owned[id] > 1));
  }).forEach(c => {
    const id = set + "-" + c.number, q = owned[id] || 0;
    const li = document.createElement("li"); if (q > 0) li.className = "have";
    li.innerHTML = '<span class="num"></span><span class="nm"><span></span><small class="rar"></small></span>';
    li.children[0].textContent = "#" + c.number; li.children[1].children[0].textContent = c.name; li.children[1].children[1].textContent = RAR[c.rarity] || c.rarity;
    const minus = btn("−", () => setQty(c, id, -1)); minus.className = "qb"; minus.setAttribute("aria-label", "Remover uma cópia de " + c.name); minus.disabled = q === 0;
    const n = document.createElement("span"); n.className = "qty"; n.textContent = q;
    const plus = btn("+", () => setQty(c, id, 1)); plus.className = "qb"; plus.setAttribute("aria-label", "Adicionar uma cópia de " + c.name);
    const star = btn(wished[id] ? "★" : "☆", () => toggleWish(id)); star.className = "qb";
    star.setAttribute("aria-label", (wished[id] ? "Remover dos desejos: " : "Adicionar aos desejos: ") + c.name); star.setAttribute("aria-pressed", wished[id] ? "true" : "false");
    li.append(star, minus, n, plus); ul.append(li);
  });
  if (!ul.children.length) ul.innerHTML = "<li>Nenhuma carta encontrada com esses filtros.</li>";
}
async function setQty(c, id, d){
  const q = Math.max(0, (owned[id] || 0) + d);
  owned[id] = q; renderCards();
  const r = q === 0
    ? await db.from("collection").delete().eq("account_id", activeId).eq("card_id", id)
    : await db.from("collection").upsert({ account_id: activeId, card_id: id, quantity: q });
  if (r.error){ say($("cardsMsg"), "Não foi possível salvar: " + r.error.message); loadOwned(true); }
}
async function toggleWish(id){
  const on = !wished[id]; wished[id] = on; renderCards();
  const r = on ? await db.from("wishlist").upsert({ account_id: activeId, card_id: id })
               : await db.from("wishlist").delete().eq("account_id", activeId).eq("card_id", id);
  if (r.error){ say($("cardsMsg"), "Não foi possível salvar: " + r.error.message); loadOwned(true); }
}
$("setSel").onchange = () => loadOwned();
$("q").oninput = () => allCards && renderCards();
$("view").onchange = () => allCards && renderCards();

db.auth.onAuthStateChange(() => route());
route();
if ("serviceWorker" in navigator) navigator.serviceWorker.register("sw.js").catch(() => {});
